{{- /* The large card image of a page, as a dict of url, size (a dict of width and height, empty when unknown) and alt, or false. The first `images` entry of the front matter (a file of the page bundle, or a path or URL), else a page bundle's cover.* image, else the site's socialImage. A file of the page bundle becomes a 1200px wide jpg. A path under the site's static/ folder gives its size; a remote URL does not. */ -}} {{- $result := false }} {{- $src := "" }} {{- $res := false }} {{- with .Params.images }} {{- $src = cond (reflect.IsSlice .) (index . 0) . }} {{- with $.Resources.Get $src }}{{ $res = . }}{{ end }} {{- else }} {{- with .Resources.GetMatch "cover.*" }}{{ $res = . }}{{ end }} {{- end }} {{- $alt := .Params.imageAlt | default .Title }} {{- if and $res (eq $res.ResourceType "image") (ne $res.MediaType.SubType "svg") }} {{- $img := $res.Resize "1200x jpg q85" }} {{- $result = dict "url" $img.Permalink "size" (dict "width" $img.Width "height" $img.Height) "alt" $alt }} {{- else if $src }} {{- $result = dict "url" (partial "site-url.html" $src) "size" (partial "static-image-size.html" $src) "alt" $alt }} {{- else }} {{- with site.Params.socialImage }} {{- $alt = site.Params.socialImageAlt | default site.Title }} {{- $result = dict "url" (partial "site-url.html" .) "size" (partial "static-image-size.html" .) "alt" $alt }} {{- end }} {{- end }} {{- return $result -}}
