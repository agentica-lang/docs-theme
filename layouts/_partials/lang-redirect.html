{{- /* The language redirect and the stored pick (DESIGN.md, Language selector). An inline script, first in the head so that the reader never sees the first language. An arrival from outside moves to this page in the reader's language when the page has that translation: the stored pick (localStorage "docs-lang") wins; with no pick, the first browser language the site has moves only a reader of the default language. "Outside" is a referrer that does not start with the site's root (its origin and first folder, the same for every version of the site), so a sibling docs site on the same origin counts as outside. A Back navigation never redirects. A change of the language select stores the pick and goes to the option. A site with one language has none of it. */ -}} {{- if gt (len hugo.Sites) 1 }} {{- $targets := dict }} {{- range .AllTranslations }}{{ $targets = merge $targets (dict .Language.Locale .RelPermalink) }}{{ end }} {{- $locales := slice }} {{- range hugo.Sites }}{{ $locales = $locales | append .Language.Locale }}{{ end }} {{- /* The site's root, shared by every build of the site: the origin and the first folder of baseURL. A version's build (baseURL `/dev/`) has the same root, so a switch between versions is not an arrival from outside. */ -}} {{- $base := urls.Parse site.BaseURL }} {{- $root := printf "%s://%s/" $base.Scheme $base.Host }} {{- with index (split (strings.Trim $base.Path "/") "/") 0 }}{{ $root = printf "%s%s/" $root . }}{{ end }} {{- $config := dict "lang" .Language.Locale "default" hugo.Sites.Default.Language.Locale "root" $root "languages" $locales "targets" $targets }} {{- end -}}
