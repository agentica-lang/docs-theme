{{- $dest := .dest -}} {{- $url := urls.Parse $dest -}} {{- $fragment := cond (eq $url.Fragment "") "" (printf "#%s" $url.Fragment) -}} {{- $page := false -}} {{- $href := $dest -}} {{- /* A page of a language that has no page at the link (a Spanish page with no Spanish version) goes to the default language's page, with hreflang of that language. */ -}} {{- $lang := "" -}} {{- $default := hugo.Sites.Default -}} {{- $fallback := ne site.Language.Lang $default.Language.Lang -}} {{- if or $url.IsAbs (eq $url.Path "") -}} {{- else if hasPrefix $url.Path "/" -}} {{- with site.GetPage $url.Path -}} {{- $page = . -}} {{- else with and $fallback ($default.GetPage $url.Path) -}} {{- $page = . -}} {{- $lang = $default.Language.Lang -}} {{- else -}} {{- warnf "%s: no page for link %q" $.from.Path $dest -}} {{- end -}} {{- else -}} {{- $source := partial "source-path.html" $.from -}} {{- $target := path.Join (path.Dir $source) $url.Path -}} {{- range site.Pages -}} {{- if eq (partial "source-path.html" .) $target }}{{ $page = . }}{{ end -}} {{- end -}} {{- if and (not $page) $fallback -}} {{- $logical := path.Join (path.Dir $.from.File.Path) $url.Path -}} {{- range $default.Pages -}} {{- if and .File (eq .File.Path $logical) }}{{ $page = . }}{{ $lang = $default.Language.Lang }}{{ end -}} {{- end -}} {{- end -}} {{- if not $page -}} {{- $href = "" -}} {{- if site.Params.repoPublic -}} {{- $href = printf "%s/blob/%s/%s%s" site.Params.repo site.Params.branch $target $fragment -}} {{- end -}} {{- /* Another file of the repository (LICENSE, a source file) is linked on GitHub, or is plain text on a private site; a link written as a page (`x.md`, `../x/`) must reach one. */ -}} {{- if or (strings.HasSuffix $url.Path ".md") (strings.HasSuffix $url.Path "/") -}} {{- warnf "%s: no page for link %q" $.from.Path $dest -}} {{- end -}} {{- end -}} {{- end -}} {{- return dict "page" $page "fragment" $fragment "href" $href "external" $url.IsAbs "lang" $lang -}}
